+  const current = currentSessionEpoch();

+const publishLoading = () => {

+export const startRequestLoading = (
+  request: SessionRequestConfig,
+  showLoading: boolean
+) => {
+  synchronizeSession();
+  if (showLoading) pending.add(request);
+  publishLoading();
+};

+export const finishRequestLoading = (request?: SessionRequestConfig) => {
+  synchronizeSession();
+  if (isStaleSessionRequest(request)) return;
+  if (request) pending.delete(request);
+  // Configless errors retain the previous global loading cleanup behavior.
+  else pending.clear();
+  publishLoading();
+};

    clearTimeout(timerPopup.current);
    timerPopup.current = 0;

    mounted.current = true;

      if (active && sessionEpoch === currentSessionEpoch()) {
        setWaitingForRefresh(false);
      }

    return () => {

    try {
      const payload = store.refreshToken
        ? getJWTPayload(store.refreshToken)
        : "";
      return payload &&
        typeof payload.exp === "number" &&
        Number.isFinite(payload.exp * 1000)
        ? payload.exp * 1000
        : 0;
    } catch {
      return 0;
    }
  }, [store.refreshToken]);
  const recoveryDeadline = refreshExpiresAt - 2000;

  React.useEffect(() => {
    clearAllTimeout();
    if (
      waitingForRefresh ||
      loading ||
      store.isOnLogout ||
      logoutStarted.current ||
      sessionEpoch !== currentSessionEpoch()
    )
      return;
    const remaining = recoveryDeadline - Date.now();
    if (remaining <= 0) {
      void continuLogout();
      return;

    timerPopup.current = setTimeout(() => void continuLogout(), remaining);

export const getRefreshToken = () => {
  const { store } = getHooksBase();
  const epoch = currentSessionEpoch();
  if (store.refreshToken) {
    try {
      const payload = getJWTPayload(store.refreshToken);
      if (
        payload &&
        typeof payload.exp === "number" &&
        Number.isFinite(payload.exp * 1000) &&
        payload.exp * 1000 - Date.now() > REFRESH_TOKEN_REUSE_WINDOW_MS
      ) {
        return;
      }
    } catch {
      // An unreadable expiry cannot establish that the token is reusable.
    }
  }
  if (
    refreshTokenRequest &&
    refreshTokenRequest.epoch === epoch &&
    refreshTokenRequest.accessToken === store.token &&
    !refreshTokenRequest.controller.signal.aborted
  ) {
    return refreshTokenRequest.promise;
  }

  signal.getRefreshToken?.abort();
  const controller = new AbortController();
  signal.getRefreshToken = controller;
  const promise = service
    .post(
      getEndPoint("/auth/v2/sso/refreshtoken"),
      { singleUIAuthorization: store.token },
      {
        signal: controller.signal,
      }
    )
    .then(() => undefined)
    .catch((e) => {
      console.error("e", e);
    })
    .finally(() => {
      if (refreshTokenRequest?.controller === controller) {
        refreshTokenRequest = undefined;
      }
      if (signal.getRefreshToken === controller) {
        signal.getRefreshToken = undefined;
      }
    });
  refreshTokenRequest = {
    accessToken: store.token,
    epoch,
    controller,
    promise,
  };
  return promise;

export const errorHandler = (error: AxiosError): unknown => {
  if (
    isStaleSessionRequest(error.config) ||
    error.code === "ERR_CANCELED" ||
    error.config?.signal?.aborted
  ) {
    finishRequestLoading(error.config);
    return Promise.reject(error);
  }
  const { baseDispatch } = getHooksBase();
  if (
    error.code !== "ERR_CANCELED" &&
    !error?.config?.url?.includes("/logout") &&
    !error?.config?.url?.includes("/api/analytics/") &&
    !error?.config?.url?.includes("/api/sse/")
  ) {
    let msg = `${error?.config?.url} : ${error.message}`;
    if (
      error?.response?.data ||
      error.message === "Network Error" ||
      error.message === "Request failed with status code 401"
    ) {
      let data: any = error?.response?.data;
      msg =
        data?.message ||
        data?.errorMessage ||
        `Error happened, service is unavailable, pls check with support team: ${error?.config?.url}: API request failed`;
      if (typeof data === "string") {
        msg = data;
      }
    }
    if (msg?.includes("TOKEN_INVALID_EXPIRED")) {
      clearStorageWhenLogout(baseDispatch);
      msg = "Expired Session, please login again.";
    }
    if (msg?.includes("AuthenticationException")) {
      console.info(error?.config);
      msg =
        "Invalid username and password combination or SSO service is unavailable.";
    }
    baseDispatch({ type: ActionType.SET_ERRORMSG, data: { errorMsg: msg } });
  }
  finishRequestLoading(error.config);
  return Promise.reject({ ...error });

const pending = new Set<SessionRequestConfig>();
let epoch = currentSessionEpoch();

const synchronizeSession = () => {
  const current = currentSessionEpoch();
  if (epoch !== current) {
    pending.clear();
    epoch = current;
  }
};

const publishLoading = () => {
  getHooksBase().baseDispatch({
    type: ActionType.SET_IS_LOADING,
    data: { isLoading: pending.size > 0 },
  });
};

export const startRequestLoading = (
  request: SessionRequestConfig,
  showLoading: boolean
) => {
  synchronizeSession();
  if (showLoading) pending.add(request);
  publishLoading();
};

export const finishRequestLoading = (request?: SessionRequestConfig) => {
  synchronizeSession();
  if (isStaleSessionRequest(request)) return;
  if (request) pending.delete(request);
  // Configless errors retain the previous global loading cleanup behavior.
  else pending.clear();
  publishLoading();
};

const isStandardResponse = (response: AxiosResponse) => {
  const requestUrl = response?.config?.url ?? "";

  return (
    standardResponseURLS.some((url) => requestUrl.includes(url)) ||
    response?.config?.headers?.["standard-response"] === "true"
  );

const handleResponse = (response: AxiosResponse): any => {
  if (
    isStaleSessionRequest(response.config) ||
    response.config?.signal?.aborted
  ) {
    throw new CanceledError(
      "Discarded an ended session response",
      response.config
    );
  }
  const loginAccess =
    response.headers?.["single-ui-authorization"] ??
    response.headers?.["Single-UI-Authorization"];
  if (response.config?.url?.includes("/auth/v2/sso/relogin")) {
    try {
      const payload =
        typeof loginAccess === "string" ? getJWTPayload(loginAccess) : "";
      if (
        !payload ||
        typeof payload.exp !== "number" ||
        !Number.isFinite(payload.exp * 1000) ||
        payload.exp * 1000 <= Date.now() ||
        typeof payload.iat !== "number" ||
        !Number.isFinite(payload.iat) ||
        payload.iat <= 0 ||
        payload.exp <= payload.iat
      ) {
        throw new Error("Unusable renewal access");
      }
    } catch {
      return errorHandler(
        new AxiosError(
          "Relogin did not return usable access",
          "ERR_BAD_RESPONSE",
          response.config
        )
      );
    }
  }
  if (isFreshLogin(response.config?.url) && loginAccess) {
    // Requests started while login was pending still belong to the old session.
    invalidateSession();
    (response.config as SessionRequestConfig).sessionEpoch =
      currentSessionEpoch();
    getHooksBase().baseDispatch({
      type: ActionType.SET_REFRESH_TOKEN,
      data: { refreshToken: undefined },
    });
  }
  if (
    response?.config?.url?.includes("/api/auth/v2/sso/login") ||
    response?.config?.url?.includes("/api/auth/v3/sso/login") ||
    response?.config?.url?.includes("/api/auth/v2/sso/relogin") ||
    response?.config?.url?.includes("/api/auth/v2/sso/validate") ||
    response?.config?.url?.includes("/api/auth/v2/sso/extend")
  ) {
    handleDrawers(response);
    handleEntities(response);
    handleEntitlementsToken(response);
    handleUser(response);
    handleLogin(response);
  }
  if (response?.config?.url?.includes("/api/auth/v2/sso/refreshtoken")) {
    handleRefreshToken(response);
  }
  if (isStandardResponse(response)) {
    return handleStandardResponse(response);
  }
  return response;
};

export const successHandler = (response: AxiosResponse): any => {
  try {
    return handleResponse(response);
  } finally {
    finishRequestLoading(response.config);
  }
};

// Format time
// Formats that do not need to be converted need to be excluded
// Possible parameter formats:
//   1. 1594006198764 (No conversion is required as may conflict with price)
//   2. 2020-02-02 (No conversion is required)
//   3. 2020-06-17T03:04:13Z
//   4. 2020-06-17 03:04:13
export const formatDate = (time: any, isAccurateToDay: boolean) => {
  if (isValidationToFormateDate(time) || isAccurateToDay) {
    const realTime = isNumber(time) ? parseInt(time) : time;
    let result = "";
    if (isAccurateToDay) {
      result = dayjs(realTime).format("YYYY MMM DD");
    } else {
      result = dayjs(realTime).format("YYYY-MM-DD HH:mm:ss");
    }
    return result !== "Invalid Date" ? result : time;
  }
  return time;
};

export const formatDateToISO = (time: any, isAccurateToDay: boolean) => {
  if (isValidationToFormateDate(time) || isAccurateToDay) {
    let newTime = "";
    try {
      const realTime = isNumber(time) ? parseInt(time) : time;
      if (isAccurateToDay) {
        newTime = dayjs.utc(realTime).format("YYYY MMM DD");
      } else {
        newTime = dayjs.utc(realTime).format();
      }
    } catch (error) {
      newTime = time;
    }

    return newTime;
  }
  return time;
};
